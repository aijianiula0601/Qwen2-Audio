# SSH Config Example for Multi-Node Training
# Save this to ~/.ssh/config
# 将此配置保存到 ~/.ssh/config

# 主节点 202.168.100.178
Host node1 202.168.100.178
    HostName 202.168.100.178
    User huangjiahong.dracu
    Port 10020
    IdentityFile ~/.ssh/id_rsa
    StrictHostKeyChecking no
    UserKnownHostsFile /dev/null
    ServerAliveInterval 60
    ServerAliveCountMax 3

# 节点 202.168.100.251  
Host node2 202.168.100.251
    HostName 202.168.100.251
    User huangjiahong.dracu
    Port 10020
    IdentityFile ~/.ssh/id_rsa
    StrictHostKeyChecking no
    UserKnownHostsFile /dev/null
    ServerAliveInterval 60
    ServerAliveCountMax 3

# 节点 202.168.100.165
Host node3 202.168.100.165
    HostName 202.168.100.165
    User huangjiahong.dracu
    Port 10020
    IdentityFile ~/.ssh/id_rsa
    StrictHostKeyChecking no
    UserKnownHostsFile /dev/null
    ServerAliveInterval 60
    ServerAliveCountMax 3

# 通用配置 - 所有 202.168.100.* 主机
Host 202.168.100.*
    User huangjiahong.dracu
    Port 10020
    StrictHostKeyChecking no
    UserKnownHostsFile /dev/null
    PasswordAuthentication no
    IdentityFile ~/.ssh/id_rsa
    ServerAliveInterval 60
    ServerAliveCountMax 3

# 说明：
# - Port 10020: 使用自定义SSH端口
# - StrictHostKeyChecking no: 跳过主机密钥检查
# - UserKnownHostsFile /dev/null: 不保存主机密钥
# - ServerAliveInterval/CountMax: 保持连接活跃
